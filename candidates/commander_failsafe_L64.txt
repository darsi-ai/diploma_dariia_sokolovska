		Mission = (1 << 0),
		AutoModes = (1 << 1),
		Offboard = (1 << 2),
		ExternalMode = (1 << 3),
		AltitudeCruise = (1 << 4)
	};

		Return = 1,         // Return mode (deprecated)

	};
