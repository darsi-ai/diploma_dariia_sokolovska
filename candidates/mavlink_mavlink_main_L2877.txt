					if (_mode == MAVLINK_MODE_IRIDIUM) {
						if (command_ack.from_external) {
							// for MAVLINK_MODE_IRIDIUM send only if external
							mavlink_msg_command_ack_send_struct(get_channel(), &msg);
						}

					} else {
						mavlink_msg_command_ack_send_struct(get_channel(), &msg);
					}

				}
			}
		}
	}
}
