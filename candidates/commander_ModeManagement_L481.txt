	if (_modes.valid(nav_state)) {
		const Modes::Mode &mode = _modes.mode(nav_state);

		if (mode.replaces_nav_state != Modes::Mode::REPLACES_NAV_STATE_NONE) {
			return mode.replaces_nav_state;
		}
	}

	return nav_state;
}
